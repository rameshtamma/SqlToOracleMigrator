# DB Conversion Master Template (Template-Driven Migration) — v1
# -------------------------------------------------------------------------
# Purpose
#   A general, "master script" template that drives a
#   conversion/migration run inside a Migration UI.
#
# How to use
#   1) Copy this file and set the placeholders in the [PLACEHOLDERS] section.
#   2) For each Stage, fill in the action scripts or connect them to tool actions.
#   3) The Migration UI reads this template and renders stages as steps.
#   4) The tool produces:
#        - Convert_<SRC>_To_<TGT>.log (mirrored stage log)
#        - ddl_validation_report.json (optional)
#        - data_validation_report.json (optional)
#        - master_generated.sql (optional, if tool compiles a runnable script)
#
# Notes
#   - Not intended to be identical; it provides the same “stage prompt + ordered actions” model.
#   - Keep scripts idempotent when possible (CREATE IF NOT EXISTS / handle already-exists).
#
# -------------------------------------------------------------------------
# [PLACEHOLDERS]  (Fill these before running)
# -------------------------------------------------------------------------
# Required
#   {{SRC_DBMS}}             e.g., SQLSERVER | POSTGRES | MYSQL | ORACLE | DB2
#   {{TGT_DBMS}}             e.g., ORACLE | SQLSERVER | POSTGRES | MYSQL | DB2
#   {{SRC_DB_NAME}}          e.g., AdventureWorks2025
#   {{TGT_DB_NAME}}          e.g., AdventureWorks2025_Orc  (or same as source)
#   {{RUN_ID}}               unique run id, e.g. run_20260105_101500
#   {{RUN_TAG}}              short label, e.g. DevLane | UAT | ProdDryRun
#   {{LOG_DIR}}              output folder, e.g. Logs/20260105/{{RUN_ID}}
#
# Connections (referenced by the tool)
#   {{SRC_CONN}}             name of saved source connection in your app
#   {{TGT_ADMIN_CONN}}       name of saved target admin connection in your app
#   {{TGT_WORK_CONN}}        optional: work connection if different from admin
#
# Security / schema strategy
#   {{SCHEMA_STRATEGY}}      SINGLE_SCHEMA | MULTI_SCHEMA_CLONE
#   {{TGT_SCHEMA}}           used only when SCHEMA_STRATEGY=SINGLE_SCHEMA
#   {{AUTO_CREATE_SCHEMAS}}  true|false (if MULTI_SCHEMA_CLONE, auto-provision users/schemas)
#   {{ROLE_NAME}}            e.g., ROLE_CONVERT_{{SRC_DBMS}}_TO_{{TGT_DBMS}}
#
# Validation settings
#   {{ENABLE_DDL_VALIDATION}}    true|false  (Data Def Validation stage)
#   {{ENABLE_DATA_VALIDATION}}   true|false  (Data Validation stage)
#   {{DATA_VALIDATE_ROW_LIMIT}}  integer (default 5000). Use 0 for ALL ROWS
#   {{VALIDATE_ALL_ON_DEMAND}}   true|false (UI can override to run all rows)
#
# Dependency/ordering knobs
#   {{ORDER_TABLES_BY_FK}}     true|false
#   {{ORDER_CONSTRAINTS}}      PK -> CHECK -> INDEX -> FK  (default)
#
# Data load knobs
#   {{DOP}}                    degree of parallelism, e.g. 4
#   {{BATCH_SIZE}}             bulk batch size, e.g. 2000
#
# -------------------------------------------------------------------------
# [EXECUTION ENVIRONMENT] (Optional; can be treated as comments by the tool)
# -------------------------------------------------------------------------
# If your tool compiles this template into a runnable script (e.g., SQL*Plus / psql),
# keep these directives. Otherwise, they serve as stage metadata for the UI.

  PROCEDURE LOG(p_level IN VARCHAR2, p_msg IN VARCHAR2);

  PROCEDURE IDENTITY_RESET(p_key IN VARCHAR2);

  -- Safe cast

  -- DDL parse/validation (used by Data Def Validation)

END UTILS;
/

CREATE OR REPLACE PACKAGE BODY UTILS AS
  PROCEDURE LOG(p_level IN VARCHAR2, p_msg IN VARCHAR2) IS
  BEGIN
    DBMS_OUTPUT.PUT_LINE('['||p_level||'] '||p_msg);
  END;

  PROCEDURE HANDLE_ERROR(p_code IN NUMBER, p_msg IN VARCHAR2, p_stage IN VARCHAR2, p_object IN VARCHAR2) IS
  BEGIN
    DBMS_OUTPUT.PUT_LINE('[ERROR] stage='||p_stage||' obj='||p_object||' code='||p_code||' msg='||p_msg);
  END;

  FUNCTION DATEADD(p_datepart IN VARCHAR2, p_number IN NUMBER, p_date IN DATE) RETURN DATE IS
  BEGIN
    -- Placeholder: implement as needed per source dialect
    RETURN p_date + p_number;
  END;

  FUNCTION DATEDIFF(p_datepart IN VARCHAR2, p_start IN DATE, p_end IN DATE) RETURN NUMBER IS
  BEGIN
    RETURN (p_end - p_start);
  END;

  FUNCTION IDENTITY(p_key IN VARCHAR2, p_seed IN NUMBER DEFAULT 1, p_increment IN NUMBER DEFAULT 1) RETURN NUMBER IS
  BEGIN
    -- Placeholder: implement with sequences / identity columns as appropriate
    RETURN p_seed;
  END;

  PROCEDURE IDENTITY_RESET(p_key IN VARCHAR2) IS
  BEGIN
    NULL;
  END;

  FUNCTION TRY_CAST_NUMBER(p_str IN VARCHAR2) RETURN NUMBER IS
    v NUMBER;
  BEGIN
    v := TO_NUMBER(p_str);
    RETURN v;
  EXCEPTION WHEN OTHERS THEN
    RETURN NULL;
  END;

  PROCEDURE PARSE_DDL(p_ddl IN CLOB) IS
    c INTEGER;
  BEGIN
    c := DBMS_SQL.OPEN_CURSOR;
    DBMS_SQL.PARSE(c, p_ddl, DBMS_SQL.NATIVE);
    DBMS_SQL.CLOSE_CURSOR(c);
  EXCEPTION WHEN OTHERS THEN
    IF DBMS_SQL.IS_OPEN(c) THEN DBMS_SQL.CLOSE_CURSOR(c); END IF;
    RAISE;
  END;
END UTILS;

#   - tool_call: invokes internal tool capability (preferred for portability)

#   - assert: fail stage if condition not met
#
# -------------------------------------------------------------------------
